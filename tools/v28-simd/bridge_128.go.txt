//go:build goexperiment.simd && (arm64 || wasm)

package bridge

import "simd/archsimd"

func (x Mask8x16) ToBits(dst []uint) int {
	dst[0] = uint(archsimd.Mask8x16(x).ToBits())
	return 1
}
