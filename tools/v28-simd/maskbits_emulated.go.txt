package simd

func (x Mask8s) ToBits(dst []uint) int {
	var word uint
	for i := 0; i < 8; i++ {
		word |= uint((x.a>>(8*i))&1) << i
		word |= uint((x.b>>(8*i))&1) << (i + 8)
	}
	dst[0] = word
	return 1
}
